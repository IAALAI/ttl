button_baf_c_c_danube_c = {
	name = button_baf_c_c_danube_c
	desc = ttl
	visible = {
		NOT = {
			exists = c:KUK
		}
	}
	possible =  {
		NOT = {
			exists = c:KUK
		}
		any_state_in_danube = {
			OR = {
				owner = root
				owner = {
					is_subject_of = root
				}
			}
			count >= 4
		}
	}
	ai_chance = {
		value = 50
	}
	effect = {
		if = {
			limit = {
				any_country = {
					is_subject_of = ROOT
					capital = {
						OR = {
							is_in_geographic_region = geographic_region_danube
						}
					}
					NOT = {
						has_diplomatic_pact = {
							who = ROOT
							type = personal_union
							is_initiator = no
						}
					}
					is_player = yes
				}
			}
			random_country = {
				limit = {
					is_subject_of = ROOT
					capital = {
						OR = {
							is_in_geographic_region = geographic_region_north_germany_old
							is_in_geographic_region = geographic_region_south_germany_old
							is_in_geographic_region = geographic_region_rhine_old
						}
					}
					NOT = {
						has_diplomatic_pact = {
							who = ROOT
							type = personal_union
							is_initiator = no
						}
					}
					is_player = yes
				}
				change_tag = RHN
			}
		}
		else_if = {
			limit = {
				any_country = {
					is_subject_of = ROOT
					capital = {
						OR = {
							is_in_geographic_region = geographic_region_north_germany_old
							is_in_geographic_region = geographic_region_south_germany_old
							is_in_geographic_region = geographic_region_rhine_old
						}
					}
					NOT = {
						has_diplomatic_pact = {
							who = ROOT
							type = personal_union
							is_initiator = no
						}
					}
				}
			}
			ordered_country = {
				limit = {
					is_subject_of = ROOT
					capital = {
						OR = {
							is_in_geographic_region = geographic_region_north_germany_old
							is_in_geographic_region = geographic_region_south_germany_old
							is_in_geographic_region = geographic_region_rhine_old
						}
					}
					NOT = {
						has_diplomatic_pact = {
							who = ROOT
							type = personal_union
							is_initiator = no
						}
					}
				}
				order_by = country_rank
				position = 0
				check_range_bounds = no
				change_tag = RHN
			}
		}
		else = {
			ordered_scope_state = {
				limit = {
					state_region = {
						OR = {
							is_homeland = cu:south_german
							is_homeland = cu:north_german
						}
					}
					OR = {
						is_in_geographic_region = geographic_region_north_germany_old
						is_in_geographic_region = geographic_region_south_germany_old
						is_in_geographic_region = geographic_region_rhine_old
					}
					is_incorporated = no
					trigger_if = {
						limit = {
							ROOT = {
								country_has_primary_culture = cu:french
							}
						}
						NOR = {
							state_region = s:STATE_RHINELAND
							state_region = s:STATE_NORTH_RHINE
							state_region = s:STATE_WALLONIA
							state_region = s:STATE_FLANDERS
						}
					}
				}
				order_by = gdp
				position = 0
				check_range_bounds = no
				save_scope_as = rhine_state
			}
			create_country = {
				tag = RHN
				origin = ROOT
				state = scope:rhine_state
			}
			ROOT = {
				create_diplomatic_pact = {
					country = c:RHN
					type = puppet
				}
			}
		}
		c:RHN ?= {
			save_scope_as = rhine_scope
		}
	}
}
