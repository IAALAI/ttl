je_baf_conquer_group = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_baf_conquer
	should_be_pinned_by_default_involved = yes
	is_shown_when_inactive = {
		root = c:BAF
	}

	possible = {
		root = c:BAF
	}

	scripted_button = button_baf_annex_rhn
	scripted_button = button_baf_annex_kuk
	scripted_button = button_baf_c_na_recovered
	scripted_button = button_baf_annex_bic
	scripted_button = button_baf_annex_southeast

	immediate = {
		if = {
			limit = {
				NOT = { has_variable = baf_conquer_progress_var }
			}
			set_variable = {
				name = baf_conquer_progress_var
				value = 0
			}
		}
		# 法兰西特色资本家
		ig:ig_industrialists = {
			set_ig_trait = ig_trait:ig_trait_tax_avoidance
			set_ig_trait = ig_trait:ig_trait_engines_of_progress
			set_ig_trait = ig_trait:ig_trait_ventilate_unify_beautify
		}
		# 法兰西特色军队
		ig:ig_armed_forces = {
			set_ig_trait = ig_trait:ig_trait_materiel_waste
			set_ig_trait = ig_trait:ig_trait_veteran_consultation
			set_ig_trait = ig_trait:ig_trait_elan_vital
		}
		# 不列颠特色小市民
		ig:ig_petty_bourgeoisie = {
			set_ig_trait = ig_trait:ig_trait_bah_humbug
			set_ig_trait = ig_trait:ig_trait_civil_service
			set_ig_trait = ig_trait:ig_trait_old_lady_of_threadneedle_street
		}
		# 不列颠特色教会（圣公会 + 新教 work_ethic）
		ig:ig_devout = {
			set_interest_group_name = ig_anglican_church
			set_ig_trait = ig_trait:ig_trait_pious_fiction
			set_ig_trait = ig_trait:ig_trait_divine_right
			set_ig_trait = ig_trait:ig_trait_work_ethic
		}
		baf_count_greater_persia = yes
	}

	# 每完成一个征服子日志 +1，凑齐 7 个后父日志完成
	current_value = {
		value = var:baf_conquer_progress_var
	}

	goal_add_value = {
		add = 7
	}

	progressbar = yes

	complete = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
	}
}

		first_valid = {
			triggered_desc = {
				desc = je_baf_conquer_middle_east_status
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_baf_persia_controlled_tt
			any_state_in_greater_persia = {
				OR = {
					owner = ROOT
					owner = { top_overlord ?= ROOT }
				}
				count = all
			}
		}
		custom_tooltip = {
			text = je_baf_russia_expelled_tt
			NOT = {
				any_state_in_greater_persia = {
					OR = {
						AND = {
							exists = c:RUS
							owner = c:RUS
						}
						AND = {
							exists = c:RUS
							owner = { top_overlord ?= c:RUS }
						}
					}
				}
			}
		}
	}

	on_complete = {
		set_variable = conquer_middle_east
